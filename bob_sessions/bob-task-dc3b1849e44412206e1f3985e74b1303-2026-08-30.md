import { Octokit } from '@octokit/rest';
import { GitHubRepoMetadata } from '../types/index.js';

export function createOctokit(token: string): Octokit {
  return new Octokit({ auth: token });
}

/**
 * Fetch metadata for all repos in an org, plus the first 500 chars of each README.
 * Pagination is handled automatically via Octokit's paginate helper.
 */
export async function fetchOrgRepoMetadata(
  octokit: Octokit,
  org: string
): Promise<GitHubRepoMetadata[]> {
  const repos = await octokit.paginate(octokit.repos.listForOrg, {
    org,
    per_page: 100,
    type: 'all',
  });

  const results: GitHubRepoMetadata[] = await Promise.all(
    repos.map(async (repo) => {
      // Ensure topics are fetched (requires Accept header for some API versions)
      let topics: string[] = repo.topics ?? [];

      let readme_excerpt: string | null = null;
      try {
        const { data: readmeData } = await octokit.repos.getReadme({
          owner: org,
          repo: repo.name,
          mediaType: { format: 'raw' },
        });
        // readmeData is a string when mediaType.format is 'raw'
        const raw = typeof readmeData === 'string' ? readmeData : String(readmeData);
        readme_excerpt = raw.slice(0, 500);
      } catch {
        // Repo has no README — acceptable, leave null
      }

      return {
        name: repo.name,
        full_name: repo.full_name,
        description: repo.description ?? null,
        topics,
        readme_excerpt,
        pushed_at: repo.pushed_at ?? null,
      };
    })
  );

  return results;
}

import { GitHubRepoMetadata, RepositoryEntry, RouterLLMResponse, RoutingResult } from '../types/index.js';

// ---------------------------------------------------------------------------
// Router system prompt — patch §4, under 200 tokens
// ---------------------------------------------------------------------------

export const ROUTER_SYSTEM_PROMPT = `You are a repository router. Given a support request and a list of repositories, select the one repository most likely to contain the relevant code.

Each repository entry has: name, description, topics, readme_excerpt.

You must return a JSON object with exactly these fields:
  selected_repo: string (full_name, e.g. "org/repo") or null
  confidence: "HIGH" | "LOW" | "AMBIGUOUS"
  reason: string (one sentence, max 20 words)

Rules:
- Use only name, description, topics, and readme_excerpt. Never request file contents.
- If no repository is a clear fit, set selected_repo to null and confidence to "AMBIGUOUS".
- Do not guess. Ambiguous is always correct when uncertain.
- Return only the JSON object. No other text.`;

// ---------------------------------------------------------------------------
// LLM call abstraction — callers inject an llmComplete function so this
// module stays testable without a real LLM (matches LLMClient pattern from §10)
// ---------------------------------------------------------------------------

export type LLMCompleteFn = (
  systemPrompt: string,
  userMessage: string
) => Promise<string>;

/**
 * Build the user message for the router LLM call.
 * Only exposes name, description, topics, readme_excerpt — never file contents (R1).
 */
function buildUserMessage(intent: string, repos: GitHubRepoMetadata[]): string {
  const repoList = repos
    .map((r) => ({
      name: r.name,
      full_name: r.full_name,
      description: r.description ?? '',
      topics: r.topics,
      readme_excerpt: r.readme_excerpt ?? '',
    }))
    .map((r) => JSON.stringify(r))
    .join('\n');

  return `Support request: "${intent}"\n\nRepositories:\n${repoList}`;
}

/**
 * Parse and validate the LLM response.
 * Returns null if the response is unparseable — caller treats that as AMBIGUOUS.
 */
function parseRouterResponse(raw: string): RouterLLMResponse | null {
  try {
    // Strip markdown fences if the LLM wraps the JSON
    const cleaned = raw.replace(/```(?:json)?\n?/g, '').trim();
    const parsed = JSON.parse(cleaned) as unknown;

    if (typeof parsed !== 'object' || parsed === null) return null;
    const obj = parsed as Record<string, unknown>;

    const confidence = obj['confidence'];
    if (confidence !== 'HIGH' && confidence !== 'LOW' && confidence !== 'AMBIGUOUS') return null;

    return {
      selected_repo: typeof obj['selected_repo'] === 'string' ? obj['selected_repo'] : null,
      confidence,
      reason: typeof obj['reason'] === 'string' ? obj['reason'] : '',
    };
  } catch {
    return null;
  }
}

/**
 * selecionar_repositorio — exact implementation of patch §2.
 *
 * Usable condition: exactly one RepositoryEntry has routing.selected == true
 *   AND its confidence is HIGH or LOW.
 * Ambiguous condition: zero selected, two+ selected, or single selected with AMBIGUOUS.
 */
export function selecionarRepositorio(candidates: RepositoryEntry[]): RoutingResult {
  const selected = candidates.filter((r) => r.routing?.selected === true);

  if (selected.length !== 1) {
    return { status: 'AMBIGUOUS' };
  }

  const entry = selected[0];
  const confidence = entry.routing!.confidence;

  if (confidence === 'AMBIGUOUS') {
    return { status: 'AMBIGUOUS' };
  }

  return { status: 'SELECTED', repo: entry, confidence };
}

/**
 * Run the router LLM call and apply its decisions back onto the entries.
 * Returns the population of RepositoryEntry objects with routing populated.
 */
export async function runRepoRouting(
  intent: string,
  githubRepos: GitHubRepoMetadata[],
  llmComplete: LLMCompleteFn
): Promise<RepositoryEntry[]> {
  const userMessage = buildUserMessage(intent, githubRepos);
  const raw = await llmComplete(ROUTER_SYSTEM_PROMPT, userMessage);
  const response = parseRouterResponse(raw);

  const now = new Date().toISOString();

  // Build RepositoryEntry list, stamping routing decisions from LLM response
  return githubRepos.map((meta): RepositoryEntry => {
    const isSelected =
      response !== null &&
      response.selected_repo === meta.full_name &&
      response.confidence !== 'AMBIGUOUS';

    const confidence =
      response === null
        ? ('AMBIGUOUS' as const)
        : isSelected
          ? response.confidence
          : ('AMBIGUOUS' as const);

    return {
      // Minimal fields from GitHub API — remaining fields (owner_team, on_call_engineer, etc.)
      // are populated from the RepositoryMap at CONTEXT_RETRIEVAL stage; for routing we only
      // need what the router touched.
      id: meta.full_name,
      name: meta.name,
      full_name: meta.full_name,
      description: meta.description ?? '',
      primary_language: '',
      owner_team: '',
      on_call_engineer: '',
      signals: meta.topics,
      main_modules: [],
      last_commit_at: meta.pushed_at ?? now,
      // Patch §1 additions
      topics: meta.topics,
      readme_excerpt: meta.readme_excerpt,
      routing: {
        selected: isSelected,
        confidence,
        reason: isSelected ? (response?.reason ?? '') : '',
        routed_at: now,
      },
      // Patch §8a — written only after selection; null here
      clone_path: null,
    };
  });
}

  return path.join(reposBaseDir(), repoName);
}

/**
 * preparar_clone — exact implementation of patch §8b.
 *
 * - If the directory does not exist: git clone
 * - If it exists: git pull --ff-only
 * - Returns the resolved clone_path (absolute)
 * - Only called after selecionar_repositorio returns SELECTED.
 */
export function prepararClone(entry: RepositoryEntry): string {
  if (!entry.clone_path) {
    throw new Error(
      `clone_path is null on entry "${entry.full_name}". ` +
        'clone_path must be set before prepararClone is called.'
    );
  }

  const clonePath = path.resolve(entry.clone_path);

  if (!existsSync(clonePath)) {
    const cloneUrl = `https://github.com/${entry.full_name}`;
    execSync(`git clone "${cloneUrl}" "${clonePath}"`, { stdio: 'inherit' });
  } else {
    execSync(`git -C "${clonePath}" pull --ff-only`, { stdio: 'inherit' });
  }

  return clonePath;
}

import { Router, Request, Response } from 'express';
import { execSync } from 'child_process';
import { createOctokit, fetchOrgRepoMetadata } from '../context/github-metadata.js';
import { runRepoRouting, selecionarRepositorio } from '../context/repo-routing.js';
import { prepararClone, deriveClonePath } from '../context/clone-manager.js';
import { DiagnoseRequest, DiagnoseResponse, DiagnoseStartedResponse } from '../types/index.js';
import { createLLMComplete } from '../llm/llm-client.js';

export const diagnoseRouter = Router();

diagnoseRouter.post('/', async (req: Request, res: Response): Promise<void> => {
  const body = req.body as Partial<DiagnoseRequest>;

  if (!body.request_id || !body.intent) {
    res.status(400).json({ error: 'request_id and intent are required' });
    return;
  }

  const { request_id, intent, system_hint, module_hint } = body as DiagnoseRequest;

  // Compose intent string the router sees — include hints when available
  const routingIntent = [intent, system_hint, module_hint].filter(Boolean).join(' | ');

  const githubToken = process.env['GITHUB_TOKEN'];
  const org = process.env['GITHUB_ORG'];

  if (!githubToken || !org) {
    res.status(500).json({ error: 'GITHUB_TOKEN and GITHUB_ORG must be set' });
    return;
  }

  // ── Step 1: fetch org repo metadata (R1 — never clones) ──────────────────
  const octokit = createOctokit(githubToken);
  let githubRepos;
  try {
    githubRepos = await fetchOrgRepoMetadata(octokit, org);
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err);
    res.status(502).json({ error: `GitHub API error: ${message}` });
    return;
  }

  // ── Step 2: router LLM call ───────────────────────────────────────────────
  const llmComplete = createLLMComplete();
  let candidates;
  try {
    candidates = await runRepoRouting(routingIntent, githubRepos, llmComplete);
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err);
    res.status(502).json({ error: `Router LLM error: ${message}` });
    return;
  }

  // ── Step 3: selecionar_repositorio ────────────────────────────────────────
  const routing = selecionarRepositorio(candidates);

  // ── Step 4: AMBIGUOUS → human review (R2) ────────────────────────────────
  if (routing.status === 'AMBIGUOUS') {
    const response: DiagnoseResponse = {
      request_id,
      revisao_humana_necessaria: true,
      reason:
        'Router confidence is AMBIGUOUS — no single repository was a clear fit. ' +
        'A human must identify the target repository before diagnosis can proceed.',
    };
    res.status(200).json(response);
    return;
  }

  // ── Step 5: write clone_path onto selected entry (§8a) ───────────────────
  const selectedEntry = routing.repo;
  selectedEntry.clone_path = deriveClonePath(selectedEntry.name);

  // ── Step 6: preparar_clone — owns clone lifecycle (§8b) ──────────────────
  let clonePath: string;
  try {
    clonePath = prepararClone(selectedEntry);
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err);
    res.status(500).json({ error: `Clone failed: ${message}` });
    return;
  }

  // ── Step 7: invoke bob --chat-mode=diagnostico (R3 scoped, R4 read-only) ──
  // bob is invoked scoped to the selected clone only; never across multiple repos.
  // The diagnostico mode is declared read-only in its mode config (R4).
  try {
    execSync(`bob --chat-mode=diagnostico -p "${routingIntent}"`, {
      cwd: clonePath,
      stdio: 'inherit',
      timeout: 60_000, // matches AGENT_EXECUTING timeout from 06_workflow_architecture.md
    });
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err);
    res.status(500).json({ error: `Diagnosis failed: ${message}` });
    return;
  }

  const response: DiagnoseStartedResponse = {
    request_id,
    selected_repo: selectedEntry.full_name,
    clone_path: clonePath,
    confidence: routing.confidence,
    status: 'DIAGNOSIS_STARTED',
  };

  res.status(200).json(response);
});

 * Minimal LLM client abstraction — matches LLMClient pattern from 10_technical_architecture.md.
 * Configured via LLM_PROVIDER env var (openai | watsonx).

 */

import { LLMCompleteFn } from '../context/repo-routing.js';

export function createLLMComplete(): LLMCompleteFn {
  const provider = process.env['LLM_PROVIDER'] ?? 'openai';

  if (provider === 'openai') {
    return createOpenAIComplete();
  }

  throw new Error(`Unsupported LLM_PROVIDER: "${provider}". Supported: openai`);
}

function createOpenAIComplete(): LLMCompleteFn {
  return async (systemPrompt: string, userMessage: string): Promise<string> => {
    const apiKey = process.env['OPENAI_API_KEY'];
    const model = process.env['OPENAI_MODEL'] ?? 'gpt-4o';

    if (!apiKey) throw new Error('OPENAI_API_KEY is not set');

    const response = await fetch('https://api.openai.com/v1/chat/completions', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${apiKey}`,
      },
      body: JSON.stringify({
        model,
        messages: [
          { role: 'system', content: systemPrompt },
          { role: 'user', content: userMessage },
        ],
        temperature: 0,
        response_format: { type: 'json_object' },
      }),
    });

    if (!response.ok) {
      const text = await response.text();
      throw new Error(`OpenAI API error ${response.status}: ${text}`);
    }

    type OpenAIChoice = { message: { content: string } };
    const data = (await response.json()) as { choices: OpenAIChoice[] };
    return data.choices[0]!.message.content;
  };
}

- **HTTP 404** → nome da org está errado (`Hacktown-BSB` tem hífen, pode ser diferente)
